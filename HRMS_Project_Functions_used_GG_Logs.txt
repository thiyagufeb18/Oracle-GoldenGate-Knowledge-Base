EXCEPTION TABLES - DDL OF THE TABLE 
####################################

SET SERVEROUTPUT ON;
DECLARE
--  v_table_name    VARCHAR2(50) := 'EMP_TB';    -- Replace with your table name
    v_owner         VARCHAR2(50) := 'HRMS2PLUS'; -- Replace with your schema/owner name
    v_sql           CLOB;
    v_column_list   CLOB;
    v_data_type     VARCHAR2(100);
    v_null          VARCHAR2(10);
    v_tablespace    VARCHAR2(100);
BEGIN
    FOR x IN (SELECT table_name FROM dba_tables WHERE owner = v_owner AND TABLE_NAME IN 
	('HRMS2_EMP_DISPLINARY_ACTION','HRMS2_EMPLOYEE_BASIC_DTLS') )  LOOP
        --v_sql := 'CREATE TABLE ' || v_owner || '.' || regexp_replace(x.table_name, '_TB$', '')|| '_ET' || ' (' || chr(10);  ---not required
        v_column_list := '';
        v_sql := 'CREATE TABLE ' || v_owner || '.' ||x.table_name || '_ET' || ' (' || chr(10);
        FOR rec IN (SELECT column_name,data_type ||
                    CASE
                        WHEN data_type IN ('VARCHAR2','CHAR') THEN '(' || data_length || ')'
                        WHEN data_type IN ('NUMBER') THEN
                            CASE
                                WHEN data_precision IS NOT NULL AND data_scale IS NOT NULL THEN '('|| data_precision || ',' || data_scale || ')'
                                WHEN data_precision IS NOT NULL THEN '(' || data_precision || ')'
                            ELSE
                                    ''
                            END
                        ELSE
                            ''
                    END
                AS data_type,
                nullable
            FROM
                dba_tab_columns
            WHERE
                table_name = upper(x.table_name)
                AND owner = upper(v_owner)
            ORDER BY
                column_id
        ) LOOP
            v_null :=
                CASE
                    WHEN rec.nullable = 'N' THEN ' NOT NULL'
                    ELSE ''
                END;
            v_column_list := v_column_list || '    ' || rec.column_name || ' ' || rec.data_type || v_null || ',' || chr(10);
        END LOOP;
        --SELECT default_tablespace INTO v_tablespace FROM dba_users WHERE username = v_owner;
        select distinct tablespace_name into v_tablespace from dba_tables where table_name=x.table_name and owner=v_owner;
        v_sql := v_sql || v_column_list;
        v_sql := v_sql || '    OPERATION VARCHAR2(5),' || chr(10);
        v_sql := v_sql || '    ERROR_CODE VARCHAR2(5),' || chr(10);
        v_sql := v_sql || '    COMMIT_TIMESTAMP DATE,' || chr(10);
        v_sql := v_sql || '    TIMESTAMP DATE' || chr(10);
        v_sql := v_sql || ') tablespace EXT_TBS;';        ------------TBS NAME
        dbms_output.put_line(v_sql);
    END LOOP;
END;
